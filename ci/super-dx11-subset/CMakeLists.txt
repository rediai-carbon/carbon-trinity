cmake_minimum_required(VERSION 3.20)
project(TrinityDx11SubsetCI LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(carbon-core CONFIG REQUIRED)
find_package(PNG CONFIG REQUIRED)
find_package(libjpeg-turbo CONFIG REQUIRED)
find_package(zlib REQUIRED)

if(NOT DEFINED CARBON_IMAGEIO_SOURCE_DIR)
    message(FATAL_ERROR "CARBON_IMAGEIO_SOURCE_DIR is required")
endif()
if(NOT DEFINED CARBON_TRINITY_SOURCE_DIR)
    message(FATAL_ERROR "CARBON_TRINITY_SOURCE_DIR is required")
endif()
set(CARBON_IMAGEIO_WITH_NANOVDB OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory("${CARBON_IMAGEIO_SOURCE_DIR}" imageio)

set(BUILD_DX11 ON CACHE BOOL "" FORCE)
set(BUILD_DX12 OFF CACHE BOOL "" FORCE)
set(BUILD_METAL OFF CACHE BOOL "" FORCE)
set(BUILD_SHADER_COMPILER OFF CACHE BOOL "" FORCE)
set(WITH_GRANNY OFF CACHE BOOL "" FORCE)
set(CARBON_TRINITY_AL_ONLY ON CACHE BOOL "" FORCE)
set(TRINITY_AL_VENDOR_FEATURES OFF CACHE BOOL "" FORCE)
set(BUILD_DOCUMENTATION OFF CACHE BOOL "" FORCE)
add_subdirectory("${CARBON_TRINITY_SOURCE_DIR}" trinity)
